  <script>
    document.addEventListener("DOMContentLoaded", () => {
      // 1. Set Current Year in Footer
      const yearEl = document.getElementById("currentYear");
      if (yearEl) {
        yearEl.textContent = new Date().getFullYear();
      }

      // 2. Mobile Navigation Toggle
      const mobileToggle = document.getElementById("mobileToggle");
      const navMenu = document.getElementById("navMenu");
      const navLinks = document.querySelectorAll(".nav-link, .nav-cta");

      mobileToggle.addEventListener("click", () => {
        const isOpen = navMenu.classList.toggle("open");
        mobileToggle.setAttribute("aria-expanded", String(isOpen));
      });

      // Close mobile menu when a link is clicked
      navLinks.forEach((link) => {
        link.addEventListener("click", () => {
          navMenu.classList.remove("open");
          mobileToggle.setAttribute("aria-expanded", "false");
        });
      });

      // 3. Navbar Shadow & Active Link Highlight on Scroll
      const navbar = document.getElementById("navbar");
      const sections = document.querySelectorAll("section[id]");

      const handleScroll = () => {
        if (window.scrollY > 20) {
          navbar.classList.add("scrolled");
        } else {
          navbar.classList.remove("scrolled");
        }

        let currentSectionId = "";
        sections.forEach((section) => {
          const sectionTop = section.offsetTop - 130;
          if (window.scrollY >= sectionTop) {
            currentSectionId = section.getAttribute("id");
          }
        });

        document.querySelectorAll(".nav-link").forEach((link) => {
          link.classList.remove("active");
          if (link.getAttribute("href") === `#${currentSectionId}`) {
            link.classList.add("active");
          }
        });
      };

      window.addEventListener("scroll", handleScroll, { passive: true });

      // 4. Pre-select Service Dropdown when clicking "Enquire Now" on Service Cards
      const serviceSelect = document.getElementById("serviceSelect");
      document.querySelectorAll(".service-link[data-service]").forEach((btn) => {
        btn.addEventListener("click", () => {
          const chosenService = btn.getAttribute("data-service");
          if (serviceSelect && chosenService) {
            serviceSelect.value = chosenService;
          }
        });
      });

      // 5. Scroll Reveal Animation via IntersectionObserver
      const revealElements = document.querySelectorAll(".reveal");
      if ("IntersectionObserver" in window) {
        const observer = new IntersectionObserver(
          (entries, obs) => {
            entries.forEach((entry) => {
              if (entry.isIntersecting) {
                entry.target.classList.add("in-view");
                obs.unobserve(entry.target);
              }
            });
          },
          { threshold: 0.12 }
        );
        revealElements.forEach((el) => observer.observe(el));
      } else {
        revealElements.forEach((el) => el.classList.add("in-view"));
      }

      // 6. Client-Side Form Validation & Demo Confirmation
      const enquiryForm = document.getElementById("enquiryForm");
      const fullNameInput = document.getElementById("fullName");
      const emailInput = document.getElementById("emailAddress");
      const phoneInput = document.getElementById("phoneNumber");
      const messageInput = document.getElementById("messageText");

      const nameError = document.getElementById("nameError");
      const emailError = document.getElementById("emailError");
      const phoneError = document.getElementById("phoneError");
      const serviceError = document.getElementById("serviceError");
      const messageError = document.getElementById("messageError");

      const formConfirmation = document.getElementById("formConfirmation");
      const confirmationDetails = document.getElementById("confirmationDetails");

      const setFieldState = (input, errorEl, isValid) => {
        if (!isValid) {
          input.classList.add("invalid");
          errorEl.classList.add("visible");
        } else {
          input.classList.remove("invalid");
          errorEl.classList.remove("visible");
        }
        return isValid;
      };

      const validateForm = () => {
        const nameVal = fullNameInput.value.trim();
        const emailVal = emailInput.value.trim();
        const phoneDigits = phoneInput.value.replace(/\D/g, "");
        const serviceVal = serviceSelect.value.trim();
        const messageVal = messageInput.value.trim();

        const isNameValid = setFieldState(fullNameInput, nameError, nameVal.length >= 2);
        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        const isEmailValid = setFieldState(emailInput, emailError, emailRegex.test(emailVal));
        const isPhoneValid = setFieldState(phoneInput, phoneError, phoneDigits.length >= 10 && phoneDigits.length <= 13);
        const isServiceValid = setFieldState(serviceSelect, serviceError, serviceVal !== "");
        const isMessageValid = setFieldState(messageInput, messageError, messageVal.length >= 10);

        return isNameValid && isEmailValid && isPhoneValid && isServiceValid && isMessageValid;
      };

      // Clear individual error states live as user types/changes
      [
        [fullNameInput, nameError],
        [emailInput, emailError],
        [phoneInput, phoneError],
        [serviceSelect, serviceError],
        [messageInput, messageError]
      ].forEach(([field, err]) => {
        field.addEventListener("input", () => {
          field.classList.remove("invalid");
          err.classList.remove("visible");
        });
      });

      enquiryForm.addEventListener("submit", (event) => {
        event.preventDefault();
        formConfirmation.classList.remove("visible");

        if (validateForm()) {
          const submittedName = fullNameInput.value.trim();
          const submittedService = serviceSelect.value;

          confirmationDetails.innerHTML = `
            Thank you, <strong>${escapeHTML(submittedName)}</strong>. Your enquiry regarding <strong>${escapeHTML(submittedService)}</strong> has been validated in this interactive front-end preview.<br /><br />
            <em>Note: Because this is a fictional portfolio website created by Webora Studio with no live backend, no real message or email has been transmitted.</em>
          `;

          formConfirmation.classList.add("visible");
          enquiryForm.reset();
        }
      });

      // Helper to sanitize user input before inserting into confirmation HTML
      function escapeHTML(str) {
        const div = document.createElement("div");
        div.textContent = str;
        return div.innerHTML;
      }
    });
  </script>
